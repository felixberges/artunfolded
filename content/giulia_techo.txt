# Eje "hacia el techo".
#   auto  -> normal del plano del recorrido (aqui sale Y, la traj. es plana en Y=0).
#   x,y,z -> forzarlo, p.ej. "arco eje: 0,1,0".
arco eje: auto

# De que lado del plano cae el techo (resuelve el signo de la normal).
#   auto  -> mira hacia donde esta la masa de la malla (aqui acierta: +Y).
#   + / - -> forzarlo si entraras mirando al suelo.
arco lado: auto

# Direccion BASE de la mirada (la cabeza luego se mueve libre con el raton).
#   arriba   -> hacia la boveda (lo que quieres aqui).
#   tangente -> mirar al frente, en el sentido del avance.
#   radial   -> hacia el centro del arco.
mirada: tangente

# Mirada al cargar, en grados: yaw,pitch (sobre la base). Aqui 44 de cabeceo
# hacia arriba. Debe caber dentro de "mirada limites" (si no, se recorta).
mirada inicial: 0,44

# Topes de la cabeza libre con el raton: +-yaw, +-pitch en grados.
# Sube estos numeros si quieres mirar mas alrededor; bajalos para acotar mas.
# Con mirada: tangente, pitch 90 = mirar en perpendicular a la boveda (cenit).
# No pasar de 90: por encima la vista se da la vuelta hacia atras.
mirada limites: 60,90

# Banda del slider de ALTURA: distancia sobre el plano del recorrido hacia el
# techo, EN METROS (la escena esta a escala real). Con el modelo recortado
# (solo techo, sep 2026) la boveda arranca a ~3,35 m y la corona esta a
# ~5,8 m sobre el camino. min,max
#   · si no llegas a rozar el techo -> sube el max
#   · si atraviesas la boveda       -> baja el max
altura: 1.2,4.0

# Altura de partida (altura de paseo).
altura inicial: 1.2

# Fraccion del arco navegable: 0,1 = todo el recorrido. Recorta (p.ej. 0.05,0.95)
# si los extremos de la fotogrametria estan sucios y no quieres llegar a ellos.
avance: 0,1

# Donde arrancas en el arco (0 = un extremo, 1 = el otro).
avance inicial: 0.0

# Interpolar con Catmull-Rom entre los 27 puntos (movimiento fluido).  si | no
avance suave: si

# Campo de vision en mm (zoom de lente, con la RUEDA). Secundario: el
# acercamiento de verdad es la ALTURA, no la lente.
lente: 24

# Topes del zoom de lente en mm.  min,max
lente limites: 15,70

# Marco de controles.
#   fija   -> siempre visible y fino (lo del mockup).
#   atenua -> se desvanece al estar quieto, vuelve al moverte (max inmersion).
